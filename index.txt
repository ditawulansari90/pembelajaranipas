  <script>
    let currentStudent = "";
    let currentPage = 1;
    const totalPages = 4;

    const keyPG = { pg1: "B", pg2: "B", pg3: "B", pg4: "C", pg5: "B", pg6: "A", pg7: "B", pg8: "A", pg9: "B", pg10: "A" };

    function checkAccess() {
      const input = document.getElementById('student-name-input').value.trim();
      if (!input) return alert("Silakan masukkan nama lengkap terlebih dahulu!");

      const history = JSON.parse(localStorage.getItem('completed_students') || '[]');
      if (history.includes(input.toLowerCase())) {
        return alert("Maaf, nama ini sudah pernah menyelesaikan latihan soal. Akses hanya diperbolehkan 1 kali!");
      }

      currentStudent = input;
      document.getElementById('user-display-name').innerText = currentStudent;
      document.getElementById('screen-login').classList.add('hidden');
      document.getElementById('screen-book').classList.remove('hidden');
      updatePage();
    }

    function updatePage() {
      for (let i = 1; i <= totalPages; i++) {
        document.getElementById(`page-${i}`).classList.toggle('active', i === currentPage);
      }
      document.getElementById('page-indicator').innerText = `Halaman ${currentPage} dari ${totalPages}`;
      document.getElementById('btn-prev').disabled = currentPage === 1;
      
      if (currentPage === totalPages) {
        document.getElementById('btn-next').classList.add('hidden');
        document.getElementById('btn-quiz').classList.remove('hidden');
      } else {
        document.getElementById('btn-next').classList.remove('hidden');
        document.getElementById('btn-quiz').classList.add('hidden');
      }
    }

    function changePage(delta) {
      currentPage += delta;
      updatePage();
    }

    function startQuiz() {
      document.getElementById('screen-book').classList.add('hidden');
      document.getElementById('screen-quiz').classList.remove('hidden');
    }

    function submitQuiz() {
      // 1. Hitung Nilai Pilihan Ganda (10 Soal x 5 Poin = Maks 50 Poin)
      let scorePG = 0;
      for (let i = 1; i <= 10; i++) {
        const selected = document.querySelector(`input[name="pg${i}"]:checked`);
        if (selected && selected.value === keyPG[`pg${i}`]) {
          scorePG += 5;
        }
      }

      // 2. Rubrik Bertingkat Uraian (10 Poin / 5 Poin / 0 Poin)
      const rubrikUraian = [
        // Soal 1
        {
          groupA: ["makan", "dimakan", "pangan"],
          groupB: ["energi", "ekosistem", "makhluk hidup"]
        },
        // Soal 2
        {
          groupA: ["rumput", "padi"],
          groupB: ["mangga", "jagung", "pohon", "tumbuhan"]
        },
        // Soal 3
        {
          groupA: ["energi"],
          groupB: ["berkurang", "puncak", "sedikit", "turun"]
        },
        // Soal 4
        {
          groupA: ["tikus", "banyak", "meningkat"],
          groupB: ["padi", "habis", "rusak", "panen"]
        },
        // Soal 5
        {
          groupA: ["pohon", "tanam", "hutan"],
          groupB: ["sampah", "daur ulang", "hemat", "jaga"]
        }
      ];

      let scoreUraian = 0;

      for (let i = 1; i <= 5; i++) {
        const val = document.getElementById(`ur${i}`).value.toLowerCase().trim();
        const rule = rubrikUraian[i - 1];

        const hasGroupA = rule.groupA.some(keyword => val.includes(keyword));
        const hasGroupB = rule.groupB.some(keyword => val.includes(keyword));

        if (hasGroupA && hasGroupB) {
          scoreUraian += 10; // Poin Sempurna (Kata kunci + Kemiripan makna utuh)
        } else if (hasGroupA || hasGroupB) {
          scoreUraian += 5;  // Poin Setengah (Hanya memuat salah satu konsep)
        }
      }

      const totalScore = scorePG + scoreUraian;

      // Simpan nama agar tidak bisa re-entry dengan nama yang sama
      const history = JSON.parse(localStorage.getItem('completed_students') || '[]');
      history.push(currentStudent.toLowerCase());
      localStorage.setItem('completed_students', JSON.stringify(history));

      // Tampilkan Hasil & Sertifikat
      document.getElementById('screen-quiz').classList.add('hidden');
      document.getElementById('screen-result').classList.remove('hidden');
      document.getElementById('final-score').innerText = totalScore;
      document.getElementById('cert-score').innerText = totalScore;
      document.getElementById('cert-name').innerText = currentStudent;
    }

    function downloadPDF() {
      const element = document.getElementById('certificate-content');
      const opt = {
        margin: 0.5,
        filename: `Sertifikat_IPAS_${currentStudent}.pdf`,
        image: { type: 'jpeg', quality: 0.98 },
        html2canvas: { scale: 2 },
        jsPDF: { unit: 'in', format: 'letter', orientation: 'portrait' }
      };
      html2pdf().set(opt).from(element).save();
    }
